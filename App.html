document.getElementById(`view-${t}`).classList.toggle('hidden', t !== tabName);

const btn = document.getElementById(`tab-${t}`);

if (t === tabName) {

btn.classList.add('tab-active');

btn.classList.remove('text-rose-500');

} else {

btn.classList.remove('tab-active');

btn.classList.add('text-rose-500');

}

});

}

 

function toggleModal(id) {

const el = document.getElementById(id);

el.classList.toggle('hidden');

if (id === 'settingsModal') {

document.getElementById('gasUrlInput').value = gasUrl;

}

}

 

function toggleInstallmentField() {

const type = document.getElementById('cfType').value;

document.getElementById('cfMonthsBox').classList.toggle('hidden', type !== '員工無息分期');

}

 

// 計算目前進行中的員工無息分期「本期應扣總額」

function getActiveInstallmentMonthlyDue() {

let totalDue = 0;

appState.cashFlows.forEach(item => {

if (item.類型 === '員工無息分期' && item.狀態 === '扣款中' && Number(item.已扣期數) < Number(item.分期總月數)) {

totalDue += Math.round(Number(item.總金額) / Number(item.分期總月數));

}

});

return totalDue;

}

 

function recalcIncomePreview() {

const gross = Number(document.getElementById('inGrossSalary').value) || 0;

const pEmp = Number(document.getElementById('pctEmp').value) || 0;

const pOp = Number(document.getElementById('pctOp').value) || 0;

const pInv = Number(document.getElementById('pctInv').value) || 0;

const pRes = Number(document.getElementById('pctRes').value) || 0;

 

document.getElementById('pctWarning').classList.toggle('hidden', (pEmp + pOp + pInv + pRes) === 100);

document.getElementById('lblEmpPct').textContent = pEmp;

 

const empGross = Math.round(gross * (pEmp / 100));

const deduct = getActiveInstallmentMonthlyDue();

const empNet = Math.max(0, empGross - deduct);

const opAmt = Math.round(gross * (pOp / 100));

const invAmt = Math.round(gross * (pInv / 100));

const resAmt = Math.round(gross * (pRes / 100)) + deduct;

 

document.getElementById('prevEmpGross').textContent = `$${empGross.toLocaleString()}`;

document.getElementById('prevInstallmentDeduct').textContent = `-$${deduct.toLocaleString()}`;

document.getElementById('prevEmpNetTransfer').textContent = `$${empNet.toLocaleString()}`;

document.getElementById('prevOpAmt').textContent = `$${opAmt.toLocaleString()}`;

document.getElementById('prevInvAmt').textContent = `$${invAmt.toLocaleString()}`;

document.getElementById('prevResAmt').textContent = `$${resAmt.toLocaleString()}`;

}

 

async function postToCloud(payload) {

if (!gasUrl) return;

try {

const res = await fetch(gasUrl, {

method: 'POST',

body: JSON.stringify(payload)

});

const json = await res.json();

if (json.status === 'success') {

appState = json.data;

saveLocal();

}

} catch (err) {

console.warn('雲端寫入失敗，已暫存本地:', err);

}

}

 

async function syncFromCloud() {

if (!gasUrl) {

alert('尚未設定 Google Sheet 網址喔！請先點右上角 ⚙️ 貼上部署網址。');

return;

}

const btn = document.getElementById('syncBtn');

btn.textContent = '⏳ 同步中...';

try {

const res = await fetch(gasUrl);

const json = await res.json();

if (json.status === 'success') {

appState = json.data;

saveLocal();

btn.textContent = '✅ 已同步';

setTimeout(() => btn.textContent = '🔄 同步雲端', 1500);

}

} catch (e) {

alert('同步失敗，請檢查網址權限是否設為「所有人」');

btn.textContent = '🔄 同步雲端';

}

}

 

function saveGasUrl() {

gasUrl = document.getElementById('gasUrlInput').value.trim();

localStorage.setItem('honeymoon_gas_url', gasUrl);

toggleModal('settingsModal');

if (gasUrl) syncFromCloud();

}

 

async function submitIncomeRecord() {

const gross = Number(document.getElementById('inGrossSalary').value) || 0;

const pEmp = Number(document.getElementById('pctEmp').value) || 0;

const pOp = Number(document.getElementById('pctOp').value) || 0;

const pInv = Number(document.getElementById('pctInv').value) || 0;

const pRes = Number(document.getElementById('pctRes').value) || 0;

 

const empGross = Math.round(gross * (pEmp / 100));

const deduct = getActiveInstallmentMonthlyDue();

const empNet = Math.max(0, empGross - deduct);

const today = new Date().toISOString().slice(0, 10);

 

const record = {

date: today,

month: today.slice(0, 7),

grossIncome: gross,

empPct: pEmp,

opPct: pOp,

invPct: pInv,

resPct: pRes,

installmentDeducted: deduct,

netEmpTransfer: empNet,

note: deduct > 0 ? `已自動內扣分期 $${deduct.toLocaleString()}` : '標準水庫分配'

};

 

// 更新進行中分期期數

for (let item of appState.cashFlows) {

if (item.類型 === '員工無息分期' && item.狀態 === '扣款中') {

item.已扣期數 = Number(item.已扣期數) + 1;

if (item.已扣期數 >= Number(item.分期總月數)) item.狀態 = '已結清';

if (gasUrl) await postToCloud({ action: 'advanceInstallment', id: item.編號 });

}

}

 

appState.incomeLogs.unshift({

日期: record.date, 月份: record.month, 入帳總薪資: record.grossIncome,

'員工比例%': record.empPct, '營運比例%': record.opPct, '投資比例%': record.invPct, '保留盈餘%': record.resPct,

本期分期內扣: record.installmentDeducted, 員工實領匯款額: record.netEmpTransfer, 備註: record.note

});

saveLocal();

await postToCloud({ action: 'addIncome', record });

alert(`💖 入帳成功！本月請直接匯款 NT$ ${empNet.toLocaleString()} 給員工羽伶！`);

}

 

async function submitAsset() {

const symbol = document.getElementById('asSymbol').value.trim().toUpperCase();

const name = document.getElementById('asName').value.trim() || symbol;

const shares = Number(document.getElementById('asShares').value) || 0;

const cost = Number(document.getElementById('asCost').value) || 0;

if (!symbol || shares <= 0) return alert('請填寫代號與股數喔！');

 

const today = new Date().toISOString().slice(0, 10);

const existing = appState.assets.find(a => a.代號 === symbol);

if (existing) {

existing.累計購入股數 = Number(existing.累計購入股數) + shares;

existing.總投入成本 = Number(existing.總投入成本) + cost;

existing.最後更新日 = today;

} else {

appState.assets.push({ 代號: symbol, 名稱: name, 累計購入股數: shares, 總投入成本: cost, 最後更新日: today });

}

saveLocal();

toggleModal('addAssetModal');

await postToCloud({ action: 'addAsset', asset: { symbol, name, shares, cost, lastDate: today } });

}

 

async function submitCashFlow() {

const type = document.getElementById('cfType').value;

const title = document.getElementById('cfTitle').value.trim();

const amount = Number(document.getElementById('cfAmount').value) || 0;

const months = type === '員工無息分期' ? (Number(document.getElementById('cfMonths').value) || 1) : 1;

if (!title || amount <= 0) return alert('請填寫項目與金額！');

 

const record = {

id: 'CF-' + Date.now().toString().slice(-4),

date: new Date().toISOString().slice(0, 10),

type, title, amount,

totalMonths: months,

paidMonths: type === '員工無息分期' ? 0 : 1,

status: type === '員工無息分期' ? '扣款中' : '已結清'

};

 

appState.cashFlows.unshift({

編號: record.id, 日期: record.date, 類型: record.type, 項目說明: record.title,

總金額: record.amount, 分期總月數: record.totalMonths, 已扣期數: record.paidMonths, 狀態: record.status

});

saveLocal();

toggleModal('addCashFlowModal');

await postToCloud({ action: 'addCashFlow', record });

}

 

async function submitAnnualReturn() {

const year = document.getElementById('yrLabel').value.trim();

const startA = Number(document.getElementById('yrStart').value) || 0;

const netIn = Number(document.getElementById('yrNetIn').value) || 0;

const endA = Number(document.getElementById('yrEnd').value) || 0;

const denom = startA + (netIn / 2);

const rate = denom > 0 ? (((endA - startA - netIn) / denom) * 100).toFixed(2) + '%' : '0.00%';

 

const record = { year, startAssets: startA, netContribution: netIn, endAssets: endA, returnRate: rate };

appState.annualReturns.unshift({

年度: year, 年初總資產: startA, 年度淨投入本金: netIn, 年底總資產: endA, '年度真實報酬率%': rate

});

saveLocal();

toggleModal('addReturnModal');

await postToCloud({ action: 'addAnnualReturn', record });

}

 

function renderAll() {

recalcIncomePreview();

 

// 計算四大水庫總累積金額

let totalInvAllocated = 0;

let totalResAllocated = 0;

appState.incomeLogs.forEach(log => {

const g = Number(log.入帳總薪資) || 0;

totalInvAllocated += Math.round(g * (Number(log['投資比例%']) / 100));

totalResAllocated += Math.round(g * (Number(log['保留盈餘%']) / 100)) + (Number(log.本期分期內扣) || 0);

});

 

// 計算買股總投入成本

let totalStockCost = 0;

appState.assets.forEach(a => totalStockCost += Number(a.總投入成本) || 0);

 

// 計算保留盈餘已流出金額

let totalResOut = 0;

let unpaidInstallments = 0;

appState.cashFlows.forEach(cf => {

totalResOut += Number(cf.總金額) || 0;

if (cf.類型 === '員工無息分期' && cf.狀態 === '扣款中') {

const perM = Math.round(Number(cf.總金額) / Number(cf.分期總月數));

const remM = Number(cf.分期總月數) - Number(cf.已扣期數);

unpaidInstallments += perM * remM;

}

});

 

const investCashLeft = totalInvAllocated - totalStockCost;

const reserveCashLeft = totalResAllocated - totalResOut;

const totalWealth = totalStockCost + Math.max(0, investCashLeft) + Math.max(0, reserveCashLeft);

 

document.getElementById('heroTotalWealth').textContent = `NT$ ${totalWealth.toLocaleString()}`;

document.getElementById('heroInvestedCost').textContent = `$${totalStockCost.toLocaleString()}`;

document.getElementById('heroCashReserve').textContent = `$${(investCashLeft + reserveCashLeft).toLocaleString()}`;

document.getElementById('bsInvestCashBalance').textContent = `$${investCashLeft.toLocaleString()}`;

document.getElementById('bsReserveCashBalance').textContent = `$${reserveCashLeft.toLocaleString()}`;

 

// 進度條（假設第一階段蜜月基金里程碑 100 萬）

const pct = Math.min(100, Math.max(5, Math.round((totalWealth / 1000000) * 100)));

document.getElementById('honeymoonProgressBar').style.width = pct + '%';

 

// 渲染損益歷史

document.getElementById('incomeHistoryList').innerHTML = appState.incomeLogs.map(l => `

<div class="bg-white/90 p-3 rounded-xl border border-pink-100 flex justify-between items-center">

<div>

<div class="font-bold text-rose-900">${l.月份} 入帳 $${Number(l.入帳總薪資).toLocaleString()}</div>

<div class="text-[11px] text-gray-400">${l.備註 || ''}</div>

</div>

<div class="text-right">

<div class="text-[11px] text-rose-500">員工實匯</div>

<div class="font-bold text-pink-600 text-sm">$${Number(l.員工實領匯款額).toLocaleString()}</div>

</div>

</div>

`).join('');

 

// 渲染股票列項（只看股數與花費）

document.getElementById('assetListContainer').innerHTML = appState.assets.map(a => `

<div class="bg-white p-3.5 rounded-xl border border-pink-100 flex justify-between items-center">

<div>

<div class="flex items-center gap-2">

<span class="font-extrabold text-rose-900 text-sm">${a.代號}</span>

<span class="text-xs text-gray-500">${a.名稱}</span>

</div>

<div class="text-[11px] text-gray-400 mt-1">累計投入成本：$${Number(a.總投入成本).toLocaleString()}</div>

</div>

<div class="text-right">

<div class="text-[11px] text-rose-400">累計股數(財富)</div>

<div class="font-extrabold text-lg text-rose-700">${Number(a.累計購入股數).toLocaleString()} <span class="text-xs font-normal">股</span></div>

</div>

</div>

`).join('');

 

// 渲染年度報酬率

document.getElementById('annualReturnList').innerHTML = appState.annualReturns.map(r => `

<div class="bg-amber-50/70 p-3 rounded-xl border border-amber-200 flex justify-between items-center">

<div>

<div class="font-bold text-amber-900">${r.年度} 年度結算</div>

<div class="text-[11px] text-amber-700">淨投入 $${Number(r.年度淨投入本金).toLocaleString()} → 年底 $${Number(r.年底總資產).toLocaleString()}</div>

</div>

<div class="font-extrabold text-base text-emerald-700">${r['年度真實報酬率%']}</div>

</div>

`).join('');

 

// 渲染內部負債

const activeInst = appState.cashFlows.filter(c => c.類型 === '員工無息分期' && c.狀態 === '扣款中');

document.getElementById('bsLiabilityList').innerHTML = activeInst.length ? activeInst.map(c => {

const perM = Math.round(Number(c.總金額) / Number(c.分期總月數));

const rem = perM * (Number(c.分期總月數) - Number(c.已扣期數));

return `

<div class="bg-amber-50 p-3 rounded-xl flex justify-between items-center">

<div>

<div class="font-bold text-amber-900">${c.項目說明}</div>

<div class="text-[11px] text-amber-700">進度：已扣 ${c.已扣期數}/${c.分期總月數} 期（每月內扣 $${perM.toLocaleString()}）</div>

</div>

<div class="font-bold text-amber-900">未還 $${rem.toLocaleString()}</div>

</div>

`;

}).join('') : '<div class="text-gray-400 text-center py-2">目前無任何員工分期欠款，零負債一身輕 ✨</div>';

 

// 渲染現金流列表

document.getElementById('cashFlowList').innerHTML = appState.cashFlows.map(c => `

<div class="bg-white p-3 rounded-xl border border-pink-100 flex justify-between items-center">

<div>

<span class="text-[10px] px-2 py-0.5 rounded-full ${c.類型 === '員工無息分期' ? 'bg-amber-100 text-amber-800' : 'bg-pink-100 text-pink-700'}">${c.類型}</span>

</div>

<div class="font-bold text-rose-800 text-sm">-$${Number(c.總金額).toLocaleString()}</div>

</div>

`).join('');

}

 

renderAll();
